/* recurse */
enum Big {
  A(u8, u16, u32, u64),
  B(i8, i16, i32, i64),
  C(bool, char, (u8, u8), (u16, u16)),
  D { x: u32, y: u32, z: u32 },
}

fn make(n: u32) -> Result<u32, Big> {
  if n > 3 { Err(Big::D { x: n, y: n, z: n }) } else { Ok(n) }
}

fn main() {
  let n = 1;
  let m = 2;
  let k = 3;
  let mut a = make(n);
  if let Err(Big::D { ref mut z, .. }) = a { *z = m; }
  for _ in 0 .. k {
    a = make(k);
  }
  let r = match a { Err(Big::D { z, .. }) => z, _ => 0 };
  `(r)`;
}
